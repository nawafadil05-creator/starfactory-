function StockTab({ stock, setStock, showToast, userRole }) {
  const [selectedItem, setSelectedItem] = useState(null);
  const [addQty, setAddQty] = useState(10);

  const handleUpdateStock = (id, delta) => {
    if (userRole === 'WORKER') {
      showToast('⚠️ Workers cannot edit stock count!', true);
      return;
    }

    setStock(prev => prev.map(s => {
      if (s.id === id) {
        const newCount = Math.max(0, s.current + delta);
        const newStatus = newCount < 20 ? 'CRITICAL' : newCount < 150 ? 'LOW' : 'OK';
        showToast(`Stock updated for ${s.name}`);
        return { ...s, current: newCount, status: newStatus };
      }
      return s;
    }));
  };

  return (
    <div className="space-y-6">
      <div className="bg-yellow-100 border-3 border-slate-900 p-4 rounded-2xl">
        <h2 className="text-xl font-black text-slate-900 uppercase">RAW MATERIAL & SAMAN COUNT</h2>
        <p className="text-xs font-bold text-slate-700">Check how much material is available in the warehouse.</p>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
        {stock.map((item) => {
          const percent = Math.min(100, Math.round((item.current / item.total) * 100));
          const isCritical = item.status === 'CRITICAL' || item.status === 'LOW';

          return (
            <div key={item.id} className="p-6 rounded-3xl border-4 border-slate-900 bg-white space-y-4 shadow-md">
              <div className="flex items-start justify-between">
                <div>
                  <span className="text-xs font-black bg-slate-900 text-yellow-400 px-2.5 py-1 rounded-lg">
                    {item.id}
                  </span>
                  <h3 className="text-lg font-black text-slate-900 mt-2">{item.name}</h3>
                </div>
                <span className={`px-3 py-1 rounded-xl border-2 border-slate-900 font-black text-xs uppercase ${
                  isCritical ? 'bg-red-500 text-white' : 'bg-emerald-400 text-slate-900'
                }`}>
                  {item.status}
                </span>
              </div>

              {/* Progress Bar Display */}
              <div className="space-y-1.5">
                <div className="flex justify-between text-xs font-black text-slate-900">
                  <span>AVAILABLE: {item.current} {item.unit}</span>
                  <span>MAX: {item.total}</span>
                </div>
                <div className="w-full h-6 bg-slate-200 rounded-xl border-2 border-slate-900 overflow-hidden p-0.5">
                  <div
                    className={`h-full rounded-lg border border-slate-900 transition-all ${
                      percent < 25 ? 'bg-red-500' : percent < 60 ? 'bg-yellow-400' : 'bg-emerald-400'
                    }`}
                    style={{ width: `${percent}%` }}
                  />
                </div>
              </div>

              {/* Simple Quick Action Buttons */}
              <div className="flex gap-2 pt-2">
                <button
                  onClick={() => handleUpdateStock(item.id, -10)}
                  className="flex-1 py-2.5 bg-red-100 hover:bg-red-200 border-2 border-slate-900 font-black text-xs rounded-xl uppercase text-slate-900"
                >
                  - 10 USED
                </button>
                <button
                  onClick={() => handleUpdateStock(item.id, +10)}
                  className="flex-1 py-2.5 bg-emerald-100 hover:bg-emerald-200 border-2 border-slate-900 font-black text-xs rounded-xl uppercase text-slate-900"
                >
                  + 10 ADDED
                </button>
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}

function WorkersTab({ workers, setWorkers, showToast }) {
  const toggleAttendance = (id) => {
    setWorkers(prev => prev.map(w => {
      if (w.id === id) {
        const nextStatus = w.status === 'PRESENT' ? 'ABSENT' : 'PRESENT';
        const nextTime = nextStatus === 'PRESENT' ? '08:30 AM' : '-';
        showToast(`${w.name} marked as ${nextStatus}`);
        return { ...w, status: nextStatus, timeIn: nextTime };
      }
      return w;
    }));
  };

  const presentCount = workers.filter(w => w.status === 'PRESENT').length;

  return (
    <div className="space-y-6">
      <div className="bg-yellow-100 border-3 border-slate-900 p-4 rounded-2xl flex items-center justify-between">
        <div>
          <h2 className="text-xl font-black text-slate-900 uppercase">WORKER ATTENDANCE (HAZIRI)</h2>
          <p className="text-xs font-bold text-slate-700">Click on worker button to mark Present or Absent.</p>
        </div>
        <div className="bg-slate-900 text-yellow-400 px-4 py-2 rounded-2xl border-2 border-slate-900 font-black text-sm">
          PRESENT: {presentCount} / {workers.length}
        </div>
      </div>

      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
        {workers.map((w) => (
          <div
            key={w.id}
            className="p-4 rounded-2xl border-3 border-slate-900 bg-white flex items-center justify-between gap-3 shadow-sm"
          >
            <div className="flex items-center gap-3">
              <span className="text-3xl p-2 bg-yellow-200 border-2 border-slate-900 rounded-xl">{w.avatar}</span>
              <div>
                <h3 className="font-black text-slate-900 text-base">{w.name}</h3>
                <p className="text-xs font-extrabold text-slate-500">{w.role} | Time: {w.timeIn}</p>
              </div>
            </div>

            <button
              onClick={() => toggleAttendance(w.id)}
              className={`px-4 py-2.5 rounded-xl border-2 border-slate-900 font-black text-xs uppercase shadow-sm ${
                w.status === 'PRESENT' ? 'bg-emerald-400 text-slate-900' : 'bg-red-500 text-white'
              }`}
            >
              {w.status}
            </button>
          </div>
        ))}
      </div>
    </div>
  );
}

function SupabaseTab({ showToast }) {
  const [copied, setCopied] = useState(false);

  const sqlScript = `-- EASY SUPABASE DATABASE SCRIPT FOR FACTORY SYSTEM
-- Paste this inside Supabase SQL Editor

CREATE TABLE public.factory_machines (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    status TEXT DEFAULT 'STOPPED',
    speed TEXT DEFAULT '0%',
    operator TEXT,
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE public.factory_inventory (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    total_stock INT DEFAULT 100,
    current_stock INT DEFAULT 50,
    unit TEXT DEFAULT 'Pcs'
);

-- Turn on Row Level Security (RLS)
ALTER TABLE public.factory_machines ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.factory_inventory ENABLE ROW LEVEL SECURITY;

-- Allow read/write access to logged-in users
CREATE POLICY "Allow All Authenticated Users" ON public.factory_machines
    FOR ALL USING (auth.role() = 'authenticated');
`;

  const copySql = () => {
    navigator.clipboard.writeText(sqlScript);
    setCopied(true);
    showToast('SQL Script copied to clipboard!');
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <div className="space-y-6">
      <div className="bg-yellow-100 border-3 border-slate-900 p-4 rounded-2xl flex items-center justify-between">
        <div>
          <h2 className="text-xl font-black text-slate-900 uppercase">SUPABASE DATABASE SCRIPT</h2>
          <p className="text-xs font-bold text-slate-700">Copy this code and paste in your Supabase SQL Editor.</p>
        </div>

        <button
          onClick={copySql}
          className="bg-slate-900 hover:bg-slate-800 text-yellow-400 border-2 border-slate-900 font-black text-xs px-4 py-2.5 rounded-xl flex items-center gap-2 shadow-sm"
        >
          {copied ? <Check className="w-4 h-4 text-emerald-400" /> : <Copy className="w-4 h-4" />}
          {copied ? 'COPIED!' : 'COPY SQL CODE'}
        </button>
      </div>

      <pre className="bg-slate-900 text-yellow-300 font-mono text-xs p-5 rounded-2xl border-4 border-slate-900 overflow-x-auto leading-relaxed">
        {sqlScript}
      </pre>
    </div>
  );
}
